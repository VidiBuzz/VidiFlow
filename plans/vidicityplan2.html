🎥 Video Processing Pipeline

flowchart LR A[Upload] --> B[Validation] B --> C[Storage] C --> D[Encoding] D --> E[Thumbnails] D --> F[Transcription] E --> G[CDN Distribution] F --> H[Search Indexing] style A fill:#3b82f6,stroke:#fff,color:#fff style D fill:#f59e0b,stroke:#fff,color:#fff style G fill:#10b981,stroke:#fff,color:#fff
Upload Flow:
  • • Presigned URLs: Direct to R2 (100MB chunks)
  • • Validation: Format, size, virus scan
  • • Formats: MP4, MOV, WebM, AVI
  • • Max Size: 10GB per video
Encoding Pipeline:
  • • Resolution: 360p, 720p, 1080p, 4K
  • • Codecs: H.264, H.265, VP9, AV1
  • • Protocol: HLS with adaptive bitrate
  • • Processing: FFmpeg + GPU acceleration
# FFmpeg Encoding Command
ffmpeg -i input.mp4 \
-vf "scale=1920:1080:force_original_aspect_ratio=decrease" \
-c:v libx264 -preset fast -crf 23 \
-c:a aac -b:a 128k \
-f hls -hls_time 6 -hls_list_size 0 \
output_1080p.m3u8

Core Data Models
PostgreSQL Tables & Relationships

Users & Profiles

CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    keycloak_id VARCHAR(36) UNIQUE NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    username VARCHAR(50) UNIQUE NOT NULL,
    tier user_tier DEFAULT 'creator',
    profile_completed BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE profiles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,
    display_name VARCHAR(100),
    bio TEXT,
    avatar_url VARCHAR(500),
    company_name VARCHAR(200),
    industry VARCHAR(100),
    city_id UUID REFERENCES cities(id),
    website VARCHAR(500),
    social_links JSONB DEFAULT '{}',
    metadata JSONB DEFAULT '{}'
);

CREATE INDEX idx_users_keycloak ON users(keycloak_id);
CREATE INDEX idx_profiles_city ON profiles(city_id);

Videos & Media

CREATE TABLE videos (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,
    title VARCHAR(255) NOT NULL,
    description TEXT,
    status video_status DEFAULT 'processing',
    visibility visibility_type DEFAULT 'public',
    
    -- Storage
    original_url VARCHAR(500),
    storage_bucket VARCHAR(100),
    storage_path VARCHAR(500),
    
    -- Video Metadata
    duration INTEGER, -- seconds
    resolution VARCHAR(20), -- 1080p, 4K, etc.
    format VARCHAR(10), -- mp4, webm, etc.
    file_size BIGINT, -- bytes
    
    -- Processing
    encoding_status JSONB DEFAULT '{}',
    transcribed BOOLEAN DEFAULT FALSE,
    transcription TEXT,
    
    -- Engagement
    view_count INTEGER DEFAULT 0,
    like_count INTEGER DEFAULT 0,
    review_count INTEGER DEFAULT 0,
    
    -- AI Features
    embedding vector(512), -- CLIP embedding
    smart_tags TEXT[],
    ai_thumbnail_url VARCHAR(500),
    
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    published_at TIMESTAMP WITH TIME ZONE
);

CREATE TABLE video_variants (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    video_id UUID REFERENCES videos(id) ON DELETE CASCADE,
    resolution VARCHAR(20) NOT NULL,
    codec VARCHAR(20) NOT NULL,
    url VARCHAR(500) NOT NULL,
    file_size BIGINT,
    bitrate INTEGER
);

CREATE INDEX idx_videos_user ON videos(user_id);
CREATE INDEX idx_videos_status ON videos(status);
CREATE INDEX idx_videos_embedding ON videos USING ivfflat (embedding vector_cosine_ops);

Reviews & Ratings

CREATE TABLE reviews (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    video_id UUID REFERENCES videos(id) ON DELETE CASCADE,
    reviewer_id UUID REFERENCES users(id) ON DELETE SET NULL,
    
    -- Review Content
    rating INTEGER CHECK (rating >= 1 AND rating <= 5),
    title VARCHAR(255),
    content TEXT,
    
    -- Video Review (optional)
    video_url VARCHAR(500),
    video_duration INTEGER,
    
    -- Moderation
    status review_status DEFAULT 'pending',
    moderation_reason VARCHAR(255),
    moderated_at TIMESTAMP WITH TIME ZONE,
    moderated_by UUID REFERENCES users(id),
    
    -- AI Analysis
    sentiment_score FLOAT, -- -1 to 1
    toxicity_score FLOAT, -- 0 to 1
    
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_reviews_video ON reviews(video_id);
CREATE INDEX idx_reviews_status ON reviews(status);

Cities & Geospatial

CREATE TABLE cities (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    slug VARCHAR(100) UNIQUE NOT NULL,
    state_province VARCHAR(100),
    country VARCHAR(100) NOT NULL,
    country_code CHAR(2),
    
    -- Location
    coordinates GEOGRAPHY(POINT, 4326),
    timezone VARCHAR(50),
    
    -- SEO & Content
    description TEXT,
    meta_title VARCHAR(255),
    meta_description TEXT(500),
    
    -- Stats
    video_count INTEGER DEFAULT 0,
    business_count INTEGER DEFAULT 0,
    
    -- Settings
    is_active BOOLEAN DEFAULT TRUE,
    featured BOOLEAN DEFAULT FALSE,
    
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_cities_slug ON cities(slug);
CREATE INDEX idx_cities_coordinates ON cities USING GIST(coordinates);
CREATE INDEX idx_cities_featured ON cities(featured) WHERE featured = TRUE;

📊 Vidi AI Configuration

Index Schema
schema business {
  document business {
    field id type string {}
    field name type string {
      indexing: index | summary
    }
    field description type string {
      indexing: index | summary
    }
    field embedding type tensor<float>(x[512]) {
      indexing: attribute | index
      attribute {
        distance-metric: angular
      }
    }
    field category type string {
      indexing: attribute | summary
    }
    field city type string {
      indexing: attribute | summary
    }
    field coordinates type position {}
    field created_at type long {}
  }
  
  rank-profile semantic {
    inputs {
      query(embedding) tensor<float>(x[512])
    }
    first-phase {
      expression: closeness(field, embedding)
    }
  }
}
